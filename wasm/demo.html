This demo simulates the output. To compile and run actual HashLink WASM code:

Quick Start (Linux/macOS)

# Install prerequisites
git clone https://github.com/emscripten-core/emsdk.git
cd emsdk && ./emsdk install latest && ./emsdk activate latest
source ./emsdk_env.sh

# Install Haxe from https://haxe.org/download/

# Build and run
cd /path/to/hashlink
./wasm/build_wasm.sh
./wasm/run_test.sh

Quick Start (Windows)

REM Install prerequisites
git clone https://github.com/emscripten-core/emsdk.git
cd emsdk
emsdk install latest
emsdk activate latest
emsdk_env.bat

REM Install Haxe from https://haxe.org/download/

REM Build and run
cd C:\path\to\hashlink
wasm\build_wasm.bat
wasm\run_test.bat
